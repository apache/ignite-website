# Apache Ignite Website - robots.txt
# Maintained at the repository root; it is not part of the Docusaurus build
# (deploy-docusaurus-build.sh leaves it alone).
# Allow rules come before the Disallow of their section for first-match parsers;
# Google and Bing pick the longest matching rule, so the order does not matter to them.

User-Agent: *
# Documentation - only index the latest versions.
# /docs/latest (with or without the trailing slash) redirects (301) to the Ignite 2
# docs and must stay crawlable, otherwise search engines never see the redirects
# from the old URLs.
Allow: /docs/$
Allow: /docs/latest
Allow: /docs/ignite2/latest
Allow: /docs/ignite3/latest/
# Page images are emitted under the pinned version directory (imagesdir in _docs/_config.template)
Allow: /docs/ignite2/*/images/
# Search UI assets loaded by every docs page
Allow: /docs/pagefind/
Disallow: /docs/
# API Documentation - only index the latest versions.
Allow: /releases/ignite/latest/
Allow: /releases/ignite2/latest/
Allow: /releases/ignite3/latest/
Disallow: /releases/
# Search results pages
Disallow: /search
Disallow: /search?q=
# Tracking and campaign parameters
Disallow: */*_sc_token=
Disallow: */*_x_tr_sl=
Disallow: */*_x_tr_tl=
Disallow: */*_x_tr_hl=
Disallow: */*_x_tr_pto=
Disallow: */*force_isolation=
Disallow: */*Preferred=
Disallow: */*utm_
Disallow: */*placement=
Disallow: */*yhid=
Disallow: */*clid=
Disallow: */?fbclid=
Disallow: */?tpclid=
Disallow: /*?_ym_debug
Disallow: /*?calltouch_tm
Disallow: */?source=
Disallow: */?web_view=
Disallow: */?event=
Disallow: */?PageSpeed=
Disallow: */?ref=
Disallow: /?from=
Disallow: /?_ga=

# Canonical host
Host: https://ignite.apache.org
